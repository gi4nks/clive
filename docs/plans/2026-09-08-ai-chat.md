# AI Chat Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a right-side drawer with an AI chat that streams responses, sees context (current dir + selected + user-dropped files), and can propose `rename`/`organize`/`create`/`trash` actions that execute only after user confirmation.

**Architecture:** New pure-function module `src/ai/chat.ts` (types, parser, validator, prompt builder, history trim) + `useChat` hook (React state + streaming wiring to existing `streamAssistant`) + `ChatDrawer.tsx` component (React + Tailwind/DaisyUI) mounted in `App.tsx` alongside the existing AI overlay. Existing `applyPlan`, `validateRenames`, `systemGuard`, `budgetFiles`, `capabilities` are reused unchanged. No Rust changes.

**Tech Stack:** TypeScript ESM, React 19, Vite 5, Tailwind 3.4, DaisyUI 5, Vitest 4, Vercel AI SDK v7 (`streamText` via `@gi4nks/wise`), Zod 4 (only if schemas needed for actions — v1 uses hand-written union).

**Spec:** `clive/docs/specs/2026-09-08-ai-chat-design.md` — read it alongside this plan.

## Global Constraints

- **ESM imports**: use `.js` extension in relative `import` statements (TypeScript ESM pattern used everywhere in this project).
- **No new npm dependencies** — everything uses what's already in `package.json`.
- **No Rust changes**: `gateway.rs`, `main.rs`, `commands/*.rs` are untouched. Chat consumes the existing `/v1/chat/completions` streaming endpoint.
- **No changes to existing AI overlay**: `AiOverlay.tsx`, `useAi.ts`, `actions.ts` stay intact; chat is a parallel path sharing only `runtime.ts`/`safety.ts`/`budget.ts`/`apply.ts`/`systemGuard.ts`.
- **System guard always respected**: any chat-proposed destructive action (rename/move/trash) on a path where `isProtectedPath()` is true must render the card greyed with hint `"sistema"` and Apply disabled.
- **User confirmation required**: no FS mutation is applied by the chat without a click on Apply. Parser failures and invalid JSON leave the message as plain text (no cards).
- **Session-only persistence**: chat history lives in React state; navigating away from the app closes it. No `localStorage` writes for chat.
- **i18n parity**: every new user-facing string in `en.json` must have an `it.json` counterpart. The existing `i18n.test.ts` coverage guard will fail otherwise.
- **Testing**: every pure function in `chat.ts` gets unit tests in `chat.test.ts` (Vitest). React components tested with `@testing-library/react` if already installed (verify in `package.json`); otherwise structural tests only via render assertion.
- **Conventional Commits** for every commit (`feat:`, `test:`, `chore:`).
- **Never commit the whole working tree** — stage only files touched by the current task.

/**
 * Chat AI pure layer (v1). Types, action-marker parsing, action validation,
 * prompt assembly, history trimming. No React, no I/O — everything here is
 * unit-testable without mocking.
 */

export const ACTIONS_OPEN = '<<<ACTIONS>>>'
export const ACTIONS_CLOSE = '<<<END>>>'

export type ChatActionType = 'rename' | 'organize' | 'create' | 'trash'

export interface RenameAction { type: 'rename'; file: string; newName: string }
export interface OrganizeAction { type: 'organize'; file: string; dstFolder: string }
export interface CreateAction { type: 'create'; file: string; content: string }
export interface TrashAction { type: 'trash'; file: string }

export type ChatAction = RenameAction | OrganizeAction | CreateAction | TrashAction

export interface ParsedActions { text: string; actions: ChatAction[] }

const ACTION_TYPES = new Set<ChatActionType>(['rename', 'organize', 'create', 'trash'])

function isActionType(v: unknown): v is ChatActionType {
  return typeof v === 'string' && ACTION_TYPES.has(v as ChatActionType)
}

function isValidAction(a: unknown): a is ChatAction {
  if (typeof a !== 'object' || a === null) return false
  const obj = a as Record<string, unknown>
  if (!isActionType(obj.type)) return false
  switch (obj.type) {
    case 'rename':
      return typeof obj.file === 'string' && typeof obj.newName === 'string'
    case 'organize':
      return typeof obj.file === 'string' && typeof obj.dstFolder === 'string'
    case 'create':
      return typeof obj.file === 'string' && typeof obj.content === 'string'
    case 'trash':
      return typeof obj.file === 'string'
  }
}

/**
 * Given a raw assistant message, split it into visible text and validated actions.
 * Tolerant: unknown types are dropped, malformed JSON leaves the block in `text`.
 */
export function extractActions(raw: string): ParsedActions {
  const re = /<<<ACTIONS>>>([\s\S]*?)<<<END>>>/i
  const m = re.exec(raw)
  if (!m) return { text: raw, actions: [] }
  try {
    const parsed = JSON.parse(m[1].trim()) as unknown
    if (typeof parsed !== 'object' || parsed === null) return { text: raw, actions: [] }
    const arr = (parsed as { actions?: unknown }).actions
    if (!Array.isArray(arr)) return { text: raw, actions: [] }
    const actions = arr.filter(isValidAction)
    const text = raw.slice(0, m.index) + raw.slice(m.index + m[0].length)
    return { text: text.trimEnd(), actions }
  } catch {
    return { text: raw, actions: [] }
  }
}
```

import { validateNewName } from './safety.js'
import { isProtectedPath } from '../systemGuard.js'

export type ValidatedAction = { ok: true; action: ChatAction } | { ok: false; error: string }

export interface ValidateCtx {
  currentDir: string
  existingNames: string[]
  guardOn: boolean
}

function parentOf(path: string): string {
  const i = path.lastIndexOf('/')
  return i > 0 ? path.slice(0, i) : '/'
}

function guardBlocks(path: string, ctx: ValidateCtx): string | null {
  if (!ctx.guardOn) return null
  if (isProtectedPath(path)) return 'system location (guard on)'
  return null
}

export function validateChatAction(a: ChatAction, ctx: ValidateCtx): ValidatedAction {
  switch (a.type) {
    case 'rename': {
      const blocked = guardBlocks(a.file, ctx)
      if (blocked) return { ok: false, error: blocked }
      const chk = validateNewName(a.newName)
      if (!chk.ok) return { ok: false, error: chk.error ?? 'invalid name' }
      if (ctx.existingNames.includes(chk.name!)) return { ok: false, error: 'name already exists' }
      return { ok: true, action: { ...a, newName: chk.name! } }
    }
    case 'organize': {
      const srcBlocked = guardBlocks(a.file, ctx)
      if (srcBlocked) return { ok: false, error: srcBlocked }
      const chk = validateNewName(a.dstFolder)
      if (!chk.ok) return { ok: false, error: chk.error ?? 'invalid folder' }
      const dst = `${ctx.currentDir}/${chk.name}`
      const dstBlocked = guardBlocks(dst, ctx)
      if (dstBlocked) return { ok: false, error: dstBlocked }
      return { ok: true, action: { ...a, dstFolder: chk.name! } }
    }
    case 'create': {
      if (parentOf(a.file) !== ctx.currentDir) return { ok: false, error: 'target outside current dir' }
      const name = a.file.slice(a.file.lastIndexOf('/') + 1)
      const chk = validateNewName(name)
      if (!chk.ok) return { ok: false, error: chk.error ?? 'invalid name' }
      const blocked = guardBlocks(a.file, ctx)
      if (blocked) return { ok: false, error: blocked }
      return { ok: true, action: { ...a, file: `${ctx.currentDir}/${chk.name}` } }
    }
    case 'trash': {
      const blocked = guardBlocks(a.file, ctx)
      if (blocked) return { ok: false, error: blocked }
      return { ok: true, action: a }
    }
  }
}
```

export interface ChatMessage {
  role: 'user' | 'assistant'
  content: string
  actions?: ChatAction[]
}

export const MAX_HISTORY_CHARS = 12_000

export const CHAT_SYSTEM_PROMPT = [
  'You are the AI assistant of Clive, a file manager.',
  'Answer in the language the user writes in.',
  'If the user asks you to perform a file operation (rename, move, create, trash),',
  'propose it at the END of your reply using ONLY this exact format:',
  '',
  ACTIONS_OPEN,
  '{"actions":[{"type":"rename","file":"/abs/path","newName":"b.txt"}]}',
  ACTIONS_CLOSE,
  '',
  'Allowed types: rename (file,newName), organize (file,dstFolder), create (file,content), trash (file).',
  'Nessuna azione viene eseguita senza conferma esplicita dell\'utente.',
  'If the user does not ask for an operation, do not emit the block.',
].join('\n')

export function trimHistory(history: ChatMessage[], maxChars: number = MAX_HISTORY_CHARS): { kept: ChatMessage[]; truncated: boolean } {
  const size = (arr: ChatMessage[]) => arr.reduce((s, m) => s + m.content.length + 20, 0)
  if (size(history) <= maxChars) return { kept: history, truncated: false }

  const first = history[0]
  const rest = history.slice(1)
  const keptRest: ChatMessage[] = []
  let budget = maxChars - (first ? first.content.length + 20 : 0)
  // Walk from the END and accumulate; then reverse.
  for (let i = rest.length - 1; i >= 0; i--) {
    const m = rest[i]
    if (m.content.length + 20 > budget) break
    keptRest.unshift(m)
    budget -= m.content.length + 20
  }
  const kept = first ? [first, ...keptRest] : keptRest
  return { kept, truncated: kept.length < history.length }
}

export interface ChatPromptCtx {
  currentDir: string
  dirListing: string
  fileContext: string
}

export function buildChatPrompt(ctx: ChatPromptCtx, history: ChatMessage[], userMsg: string): { system: string; prompt: string } {
  const { kept } = trimHistory(history)
  const parts: string[] = [
    `Current directory: ${ctx.currentDir}`,
    ctx.dirListing ? `\nDirectory contents:\n${ctx.dirListing}` : '',
    ctx.fileContext ? `\nSelected/attached file contents:\n${ctx.fileContext}` : '',
    '',
    'Conversation so far:',
    ...kept.map((m) => `${m.role === 'user' ? 'User' : 'Assistant'}: ${m.content}`),
    `User: ${userMsg}`,
    '',
    'Assistant:',
  ]
  return { system: CHAT_SYSTEM_PROMPT, prompt: parts.filter(Boolean).join('\n') }
}

import type { FileEntry } from '../api.js'

/** Compact directory listing (names only, folders marked). Truncates past maxEntries. */
export function renderDirListing(entries: FileEntry[], maxEntries = 200): string {
  const lines = entries.slice(0, maxEntries).map((e) => (e.isDir ? `${e.name}/` : e.name))
  const remaining = entries.length - Math.min(entries.length, maxEntries)
  if (remaining > 0) lines.push(`... + ${remaining} more`)
  return lines.join('\n')
}

```ts
import { useCallback, useRef, useState } from 'react'
import { deleteToTrash, moveFile, writeTextFile, type FileEntry } from '../api.js'
import { applyPlan } from './apply.js'
import { budgetFiles, renderBudgeted, type FileContent } from './budget.js'
import { extractActions, validateChatAction, buildChatPrompt, type ChatAction, type ChatMessage } from './chat.js'
import { collectFileContents } from './fs.js'
import { streamChat } from './runtime.js'

export interface UseChatDeps {
  currentPath: string
  entries: FileEntry[]
  selected: FileEntry[]
  model: string
  guardOn: boolean
  setStatus: (msg: string, kind: 'success'|'error'|'info') => void
  refresh: () => void
}

export interface ChatMessageState extends ChatMessage {
  streaming?: boolean
  actionsApplied?: boolean[]
  actionErrors?: (string | null)[]
}

export function useChat(deps: UseChatDeps) {
  const depsRef = useRef(deps)
  depsRef.current = deps

  const [messages, setMessages] = useState<ChatMessageState[]>([])
  const [addedFiles, setAddedFiles] = useState<FileEntry[]>([])
  const [streaming, setStreaming] = useState(false)
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState('')
  const abortRef = useRef<AbortController | null>(null)

  const addFile = useCallback((f: FileEntry) => setAddedFiles((prev) => (prev.some((p) => p.path === f.path) ? prev : [...prev, f])), [])
  const removeAdded = useCallback((path: string) => setAddedFiles((prev) => prev.filter((p) => p.path !== path)), [])
  const reset = useCallback(() => { setMessages([]); setAddedFiles([]); setError('') }, [])

  const send = useCallback(async (text: string) => {
    if (!text.trim()) return
    setError('')
    const userMsg: ChatMessageState = { role: 'user', content: text }
    setMessages((prev) => [...prev, userMsg])
    const asst: ChatMessageState = { role: 'assistant', content: '', streaming: true }
    setMessages((prev) => [...prev, asst])
    setStreaming(true)
    setBusy(true)
    const ctrl = new AbortController()
    abortRef.current = ctrl
    try {
      const d = depsRef.current
      const ctx = d.selected.length || addedFiles.length
        ? renderBudgeted(budgetFiles(await collectFileContents([...d.selected, ...addedFiles])))
        : ''
      const dirListing = '' // Filled from d.entries by caller — see App.tsx integration
      const { system, prompt } = buildChatPrompt({ currentDir: d.currentPath, dirListing, fileContext: ctx }, messages, text)
      let raw = ''
      for await (const chunk of streamChat({ model: d.model, system, prompt, signal: ctrl.signal })) {
        raw += chunk
        setMessages((prev) => {
          const next = [...prev]
          const last = { ...next[next.length - 1] }
          last.content = raw
          next[next.length - 1] = last
          return next
        })
      }
      const { text: clean, actions } = extractActions(raw)
      setMessages((prev) => {
        const next = [...prev]
        const last = { ...next[next.length - 1] }
        last.content = clean
        last.actions = actions
        last.streaming = false
        if (actions.length) {
          last.actionsApplied = actions.map(() => false)
          last.actionErrors = actions.map(() => null)
        }
        next[next.length - 1] = last
        return next
      })
    } catch (e) {
      if (!ctrl.signal.aborted) setError(String(e))
      setMessages((prev) => {
        const next = [...prev]
        if (next[next.length - 1]?.streaming) {
          next[next.length - 1] = { ...next[next.length - 1], streaming: false }
        }
        return next
      })
    } finally {
      setStreaming(false)
      setBusy(false)
      abortRef.current = null
    }
  }, [addedFiles, messages])

  const applyAction = useCallback(async (msgIdx: number, actIdx: number) => {
    const d = depsRef.current
    const msg = messages[msgIdx]
    const action = msg.actions?.[actIdx]
    if (!action) return
    const v = validateChatAction(action, {
      currentDir: d.currentPath,
      existingNames: d.entries.map((e) => e.name),
      guardOn: d.guardOn,
    })
    const patch = (fn: (s: ChatMessageState) => ChatMessageState) =>
      setMessages((prev) => prev.map((m, i) => (i === msgIdx ? fn(m) : m)))

    if (!v.ok) {
      patch((m) => ({ ...m, actionErrors: [...(m.actionErrors ?? []), v.error].slice(0, actIdx + 1) }))
      d.setStatus(v.error, 'error')
      return
    }
    try {
      const a = v.action
      if (a.type === 'rename') {
        await applyPlan({
          rename: (p, n) => import('../api.js').then((m) => m.rename(p, n)),
          move: async () => { /* not used for pure rename */ },
          createFolder: async () => { throw new Error('unexpected') },
          exists: async () => true,
        }, { renames: [{ path: a.file, newName: a.newName }] })
      } else if (a.type === 'organize') {
        const folderPath = `${d.currentPath}/${a.dstFolder}`
        await applyPlan({
          rename: async () => { throw new Error('unexpected') },
          move: moveFile,
          createFolder: async (parent, name) => import('../api.js').then((m) => m.createFolder(parent, name)),
          exists: async () => true,
        }, { folders: [folderPath], moves: [{ src: a.file, dst: `${folderPath}/${a.file.slice(a.file.lastIndexOf('/') + 1)}` }] })
      } else if (a.type === 'create') {
        await writeTextFile(a.file, a.content)
      } else if (a.type === 'trash') {
        await deleteToTrash(a.file)
      }
      patch((m) => {
        const applied = [...(m.actionsApplied ?? [])]
        applied[actIdx] = true
        return { ...m, actionsApplied: applied }
      })
      d.setStatus('Applicata', 'success')
      d.refresh()
    } catch (e) {
      const msg2 = String(e)
      patch((m) => {
        const errs = [...(m.actionErrors ?? [])]
        errs[actIdx] = msg2
        return { ...m, actionErrors: errs }
      })
      d.setStatus(msg2, 'error')
    }
  }, [messages])

  const cancelAction = useCallback((msgIdx: number, actIdx: number) => {
    setMessages((prev) => prev.map((m, i) => {
      if (i !== msgIdx) return m
      const cancelled = [...(m.actionsApplied ?? [])]
      cancelled[actIdx] = true // reuse flag as "no more interactive"
      const errs = [...(m.actionErrors ?? [])]
      errs[actIdx] = 'annullata'
      return { ...m, actionsApplied: cancelled, actionErrors: errs }
    }))
  }, [])

  return {
    messages,
    streaming,
    busy,
    error,
    addedFiles,
    send,
    abort: () => abortRef.current?.abort(),
    applyAction,
    cancelAction,
    reset,
    addFile,
    removeAdded,
  }
}

export type ChatController = ReturnType<typeof useChat>
```

import { useEffect, useRef, useState } from 'react'
import { useTranslation } from 'react-i18next'
import type { ChatController } from './useChat.js'
import type { ChatAction } from './chat.js'

export function ChatDrawer({ open, onClose, ai }: { open: boolean; onClose: () => void; ai: ChatController }) {
  const { t } = useTranslation()
  const [input, setInput] = useState('')
  const endRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (!open) return
    endRef.current?.scrollIntoView({ behavior: 'smooth' })
  }, [ai.messages, open])

  useEffect(() => {
    if (!open) return
    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose() }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [open, onClose])

  if (!open) return null

  const submit = () => {
    const text = input.trim()
    if (!text) return
    setInput('')
    void ai.send(text)
  }

  return (
    <div className="fixed inset-0 z-40 flex justify-end">
      <div className="absolute inset-0 bg-black/40" onClick={onClose} aria-hidden />
      <div className="relative z-10 flex h-full w-[420px] flex-col border-l border-base-300 bg-base-100 shadow-2xl">
        <div className="flex items-center justify-between border-b border-base-300 px-4 py-3">
          <span className="text-sm font-semibold">{t('chat.title')}</span>
          <button className="btn btn-ghost btn-xs" onClick={onClose} aria-label={t('common.close')}>✕</button>
        </div>

        <div className="scroll-area flex-1 overflow-y-auto p-3 text-sm">
          {ai.error && (

              <span>⚠️</span>
              <span className="min-w-0 flex-1 break-words">{ai.error}</span>
              <button className="btn btn-ghost btn-xs !min-h-0" onClick={() => navigator.clipboard?.writeText(ai.error)}>↻</button>
            </div>
          )}
          {ai.messages.length === 0 && (

              <span className="text-3xl">💬</span>

            <div key={i} className={`mb-2 ${m.role === 'user' ? 'text-right' : ''}`}>
              <div className={`inline-block max-w-[85%] whitespace-pre-wrap break-words rounded-2xl px-3 py-2 ${m.role === 'user' ? 'bg-primary/10' : 'bg-base-200'}`}>
                {m.content}

              </div>

        </div>

              </span>
            ))}
          </div>
        )}

        <div className="flex items-end gap-2 border-t border-base-300 p-2">
          <textarea
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={(e) => { if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) { e.preventDefault(); submit() } }}

        </div>
      </div>
    </div>
  )
}

function ActionCard({ a, applied, error, onApply, onCancel }: { a: ChatAction; applied?: boolean; error?: string | null; onApply: () => void; onCancel: () => void }) {
  const { t } = useTranslation()
  const label = t(`chat.action.${a.type}`)
  const summary = describeAction(a)
  const isApplied = !!applied && !error
  const isCancelled = !!applied && error === 'annullata'
  return (
    <div className="mt-1 rounded-lg border border-base-300 bg-base-200/50 p-2 text-xs">
      <div className="mb-1 font-semibold">{iconFor(a.type)} {label}</div>
      <div className="mb-1 break-words opacity-80">{summary}</div>
      {error && !isCancelled && <div className="mb-1 text-error">{error}</div>}
      {isApplied && <div className="text-success">✓ {t('chat.applied')}</div>}
      {isCancelled && <div className="opacity-60">— {t('chat.cancelled')}</div>}
      {!applied && !error && (
        <div className="flex gap-1">
          <button className="btn btn-primary btn-xs" onClick={onApply}>{t('common.apply')}</button>
          <button className="btn btn-ghost btn-xs" onClick={onCancel}>{t('common.cancel')}</button>
        </div>
      )}
    </div>
  )
}

function iconFor(type: ChatAction['type']): string {
  return { rename: '🔄', organize: '📁', create: '✨', trash: '🗑️' }[type]
}

function describeAction(a: ChatAction): string {
  switch (a.type) {
    case 'rename': return `${a.file} → ${a.newName}`
    case 'organize': return `${a.file} → ${a.dstFolder}/`
    case 'create': return a.file
    case 'trash': return a.file
  }
}
