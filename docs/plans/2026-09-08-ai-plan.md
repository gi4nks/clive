# AI Plan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn an AI-chat reorganization analysis into an interactive, tree-previewed **Plan** the user reviews/edits, then applies atomically (create folders → rename → move → trash) with a persistent, cross-restart **Undo all**.

**Architecture:** A pure TS layer (`plan.ts`: expand the model's name-based JSON against the real directory listing into validated `PlanOp[]`) + a `generatePlan` second model call (`chat.ts`) + an authoritative Rust batch engine (`commands/plan.rs`: `apply_plan`/`undo_plan` with a per-batch on-disk inverse journal and `plan-progress` events) + a `usePlan` controller + `PlanPanel` tree UI, triggered by a 🪄 button on assistant messages in `ChatDrawer`. The Plan **coexists** with the inline `<<<ACTIONS>>>` cards; it does not replace them.

**Tech Stack:** TypeScript ESM, React 19, Vite, Tailwind/DaisyUI 5, Vitest, Vercel AI SDK v7 (`generateObject` via existing gateway), Rust (Tauri 2, `serde`, `@tauri-apps/api/event`), existing `safety.ts`/`systemGuard.ts`/`budget.ts`/`apply.ts::InverseOp`.

**Spec:** `clive/docs/specs/2026-09-08-ai-plan-design.md` — the plan argues from the spec; read both. Section refs (§N) below point into the spec.

## Global Constraints

- **ESM imports** use `.js` extension in relative paths (also `.tsx` where the file is a component).
- **No new runtime npm deps.** `@tauri-apps/api` (incl. `/event`) and markdown libs are already present.
- **No Rust changes outside `commands/plan.rs` + `main.rs` registration** (and a `mod plan;` declaration). Do not touch the gateway.
- **Authoritative guard**: `apply_plan`/`undo_plan` MUST reject any protected path when `!allowSystem`, independent of the client (spec §8.1, §14.9). Mirror the existing `allow_system` pattern in `commands/filesystem.rs` / `commands/trash.rs`.
- **Phased ordering always**: `createFolder` (depth ascending) → `rename` → `move` → `trash` (spec §8.1).
- **Journal persisted** at `~/.clive/plan-journal/<batchId>.json` (spec §5.4); `batchId` = client-generated `crypto.randomUUID()`. Each journal carries `"version": 1`.
- **Nothing applies without a click**; ops flagged with `issue` are excluded and NOT re-enableable in the panel (spec §5.2, §7).
- **Second confirm** only when the plan contains ≥1 `trash` op (spec §7).
- **Hard cap**: 🪄 disabled when the current listing exceeds **5000** entries (spec §6, §13).
- **v1 scope is flat**: operates on the current folder's immediate children only (no recursion) — spec §2.
- **Idempotent apply**: an op whose source no longer exists is `skipped`, not `failed` (spec §8.1, §10).
- **i18n parity** en↔it for every new `t()` key (coverage guard fails otherwise).
- **Conventional Commits**; stage only the current task's files; never `git add -A`.
- **Test-first**: each task's steps are RED → GREEN → commit.

```ts
import { validateNewName } from './safety.js'
import { isProtectedPath } from '../systemGuard.js'
import type { FileEntry } from '../api.js'

export interface ModelPlanShape {
  folders: string[]
  assignments: { file: string; to: string }[]
  renames: { file: string; to: string }[]
  trash: { file: string }[]
  notes?: string
}
export type PlanOp =
  | { id: string; kind: 'createFolder'; path: string; included: boolean; issue?: string }
  | { id: string; kind: 'rename'; path: string; newName: string; included: boolean; issue?: string }
  | { id: string; kind: 'move'; path: string; destDir: string; included: boolean; issue?: string }
  | { id: string; kind: 'trash'; path: string; included: boolean; issue?: string }
export interface Plan { rootPath: string; ops: PlanOp[]; unchanged: FileEntry[]; warnings: string[] }
export interface ExpandCtx { rootPath: string; entries: FileEntry[]; guardOn: boolean }

const join = (root: string, rel: string): string => {
  const clean = rel.replace(/^\/+/, '').replace(/\/+$/, '')
  return clean ? `${root.replace(/\/+$/, '')}/${clean}` : root
}

export function validateFolderPath(relPath: string): { ok: boolean; reason?: string; segments?: string[] } {
  const segments = relPath.replace(/^\/+/, '').replace(/\/+$/, '').split('/').filter((s) => s.length > 0)
  if (!segments.length || relPath.includes('//') || relPath.startsWith('/') || relPath.endsWith('/')) {
    if (!segments.length) return { ok: false, reason: 'empty' }
  }
  for (const seg of segments) {
    if (seg === '.' || seg === '..') return { ok: false, reason: 'traversal' }
    const chk = validateNewName(seg)
    if (!chk.ok) return { ok: false, reason: chk.reason }
  }
  return { ok: true, segments }
}

export function expandPlan(json: ModelPlanShape, ctx: ExpandCtx): Plan {
  const ops: PlanOp[] = []
  const warnings: string[] = []
  let seq = 0
  const nextId = (k: string) => `${k}:${seq++}`
  const byName = new Map<string, FileEntry>()
  for (const e of ctx.entries) if (!byName.has(e.name)) byName.set(e.name, e)

  // 1. folders: expand each into cumulative createFolder chain, depth asc
  const seenFolders = new Set<string>()
  const folderAdds: PlanOp[] = []
  const addFolderChain = (rel: string): string => {
    const clean = rel.replace(/^\/+/, '').replace(/\/+$/, '')
    if (!clean) return ctx.rootPath
    const parts = clean.split('/')
    let acc = ''
    for (const part of parts) {
      acc = acc ? `${acc}/${part}` : part
      if (seenFolders.has(acc)) continue
      const v = validateFolderPath(acc)
      const absPath = join(ctx.rootPath, acc)
      const issue = v.ok ? undefined : 'invalid folder path'
      seenFolders.add(acc)
      folderAdds.push({ id: nextId('createFolder'), kind: 'createFolder', path: absPath, included: v.ok, issue })
    }
    return join(ctx.rootPath, clean)
  }

  // 2. resolve a basename to a listing entry with tolerant matching.
  const resolve = (name: string): FileEntry | undefined => {
    if (byName.has(name)) return byName.get(name)
    const lower = name.toLowerCase()
    let hit = ctx.entries.find((e) => e.name.toLowerCase() === lower)
    if (hit) return hit
    const withSuffix = ctx.entries.find((e) => e.name === name || new RegExp(`^${name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}( \\(\\d+\\))?$`).test(e.name))
    return withSuffix
  }

  const guardBlock = (abs: string): string | undefined => (ctx.guardOn && isProtectedPath(abs) ? 'protected' : undefined)

  // rename ops (before moves so we can rename then place)
  for (const r of json.renames ?? []) {
    const src = resolve(r.file)
    const id = nextId('rename')
    if (!src) { ops.push({ id, kind: 'rename', path: r.file, newName: r.to, included: false, issue: 'not found' }); warnings.push(`rename unknown: ${r.file}`); continue }
    const chk = validateNewName(r.to)
    const g = guardBlock(src.path)
    const issue = !chk.ok ? 'invalid name' : g ? 'protected' : ctx.entries.some((e) => e !== src && e.name === chk.name) ? 'target exists' : undefined
    ops.push({ id, kind: 'rename', path: src.path, newName: chk.name ?? r.to, included: !issue, issue })
  }

  // assignment moves
  const assigned = new Set<string>()
  const moveTargets = new Map<string, string>() // src path -> destDir (last wins)
  for (const a of json.assignments ?? []) {
    const src = resolve(a.file)
    const id = nextId('move')
    if (!src) { ops.push({ id, kind: 'move', path: a.file, destDir: '', included: false, issue: 'not found' }); warnings.push(`assign unknown: ${a.file}`); continue }
    const destAbs = addFolderChain(a.to)
    if (destAbs === ctx.rootPath) continue // no-op move to current dir
    if (assigned.has(src.path)) { const prev = moveTargets.get(src.path); ops.forEach((o) => { if (o.kind === 'move' && o.path === src.path && o.destDir !== destAbs) { o.included = false; o.issue = 'duplicate move' } }) }
    assigned.add(src.path)
    moveTargets.set(src.path, destAbs)
    const g = guardBlock(src.path) || guardBlock(destAbs)
    ops.push({ id, kind: 'move', path: src.path, destDir: destAbs, included: !g, issue: g ? 'protected' : undefined })
  }

  // trash ops
  for (const t of json.trash ?? []) {
    const src = resolve(t.file)
    const id = nextId('trash')
    if (!src) { ops.push({ id, kind: 'trash', path: t.file, included: false, issue: 'not found' }); warnings.push(`trash unknown: ${t.file}`); continue }
    const g = guardBlock(src.path)
    ops.push({ id, kind: 'trash', path: src.path, included: !g, issue: g ? 'protected' : undefined })
  }

  // prepend folder creates, ordered by depth
  const ordered = [...folderAdds.sort((a, b) => depth(a.path) - depth(b.path))]
  const all = [...ordered, ...ops]

  const touched = new Set<string>()
  for (const o of all) if ('path' in o) touched.add(o.path)
  const unchanged = ctx.entries.filter((e) => !touched.has(e.path))

  return { rootPath: ctx.rootPath, ops: all, unchanged, warnings }
}

function depth(p: string): number {
  return p.split('/').filter(Boolean).length
}

export function orderPlanOps(ops: PlanOp[]): PlanOp[] {
  const rank = (k: PlanOp['kind']) => k === 'createFolder' ? 0 : k === 'rename' ? 1 : k === 'move' ? 2 : 3
  return [...ops].sort((a, b) => rank(a.kind) - rank(b.kind) || depth((a as any).path) - depth((b as any).path))
}
```
